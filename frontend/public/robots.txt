# Upokoron -- what search engines may crawl.
#
# Crawl: the shop itself -- home, product pages, category pages (including
# their ?page=2, ?page=3), the rewards, about, contact, privacy and terms
# pages, product images, and the XML sitemap.
#
# Do not crawl: anything private to one visitor (cart, checkout, account,
# orders, wishlist, login), the admin panel, and the endless variations of a
# listing made by search, sort and filter settings -- those are the same
# products over and over, and crawling them wastes Google's time on the site
# that should go to real product pages.

User-agent: *
Allow: /

# Admin panel and server scripts.
Disallow: /admin
Disallow: /update.php
Disallow: /sanctum/

# The API is off limits EXCEPT the shop's public data. The storefront is
# built in the browser from /api/v1/shop, and Google renders pages the same
# way -- block that and every product page reads as empty to Google.
Disallow: /api/
Allow: /api/v1/shop/

# ...but never the parts of it that belong to one customer: who they are,
# where they live, what they ordered, their cart, saved payment details,
# points and own reviews. These answer nothing without the customer's login
# or order details anyway; this makes the "not for search" explicit, and
# being longer than the Allow above, these rules win over it.
Disallow: /api/v1/shop/auth/
Disallow: /api/v1/shop/addresses
Disallow: /api/v1/shop/cart
Disallow: /api/v1/shop/checkout
Disallow: /api/v1/shop/orders
Disallow: /api/v1/shop/payment-methods
Disallow: /api/v1/shop/rewards/history
Disallow: /api/v1/shop/*/mine

# Private to one visitor.
Disallow: /cart
Disallow: /checkout
Disallow: /wishlist
Disallow: /account
Disallow: /orders
Disallow: /order-complete/
Disallow: /track
Disallow: /login
Disallow: /register
Disallow: /forgot-password
Disallow: /reset-password

# Search results and filtered or sorted copies of a listing.
Disallow: /*?*search=
Disallow: /*?*sort=
Disallow: /*?*min_price=
Disallow: /*?*max_price=
Disallow: /*?*min_rating=
Disallow: /*?*category=

Sitemap: https://upokoron.com/sitemap.xml
